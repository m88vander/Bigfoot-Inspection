// Send Email via Formspree with the PDF attached function sendEmailWithAttachment(report, pdfBlob, filename, history) { const emailBody = buildEmailBody(report); const formData = new FormData(); formData.append('email', TEST_EMAIL); formData.append('subject', `🦶 Bigfoot Daily Inspection - ${report.date} ${report.shift} Shift`); formData.append('message', emailBody); formData.append('_captcha', 'false'); formData.append('attachment', pdfBlob, filename); fetch('https://formspree.io/f/meebrjgr', { method: 'POST', body: formData }).then(async response => { if (response.ok) { console.log('Email with PDF attachment sent to', TEST_EMAIL); report.status = 'sent'; if (history && history[0]) history[0].status = 'sent'; localStorage.setItem('inspectionHistory', JSON.stringify(history)); loadHistory(); } else { const data = await response.json().catch(() => null); console.error('Formspree rejected the submission:', response.status, data); showErrorMessage(); } }).catch(error => { console.error('Error sending email:', error); showErrorMessage(); }); }
